`import requests`

`# Endpoint de Contratos Electrónicos SECOP II`  
`url = "https://www.datos.gov.co/resource/6qex-kahp.json"`

`# Filtramos usando SoQL (Socrata Query Language)`  
`# Buscamos contratos del sector 'Obras' en un departamento específico`  
`params = {`  
    `"departamento_entidad": "Meta",`  
    `"estado_contrato": "En ejecución",`  
    `"$limit": 5  # Traer solo 5 para probar`  
`}`

`# Si tuvieras tu App Token (recomendado), lo pasarías así:`  
`headers = {`  
    `"X-App-Token": "TU_TOKEN_AQUI"`  
`}`

`response = requests.get(url, params=params, headers=headers)`  
`data = response.json()`

`for contrato in data:`  
    `print(f"Obra: {contrato.get('descripcion_del_proceso')}")`  
    `print(f"Contratista: {contrato.get('nombre_proveedor')}")`
